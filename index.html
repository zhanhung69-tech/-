function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var type = data.type;
    var payload = data.data;

    if (type === "leave") {
      var sh = ss.getSheetByName("請假清單") || ss.insertSheet("請假清單");
      if (sh.getLastRow() === 0) sh.appendRow(["假單ID","班級","學生名單","假別","開始日","結束日","原因","導師審核","舍監審核","主管核准","返宿日","狀態"]);
      var rows = sh.getDataRange().getValues();
      var found = false;
      for (var i = 1; i < rows.length; i++) {
        if (rows[i][0] === payload.id) {
          sh.getRange(i+1,1,1,12).setValues([[payload.id,payload.cls,payload.names,payload.type,payload.start,payload.end,payload.reason,payload.a1,payload.a2,payload.a3,payload.ret,payload.status]]);
          found = true; break;
        }
      }
      if (!found) sh.appendRow([payload.id,payload.cls,payload.names,payload.type,payload.start,payload.end,payload.reason,payload.a1,payload.a2,payload.a3,payload.ret,payload.status]);
    }

    if (type === "score") {
      var sh = ss.getSheetByName("內務評分") || ss.insertSheet("內務評分");
      if (sh.getLastRow() === 0) sh.appendRow(["日期","棟別","房號","檢查人","地板","桌面","床鋪","垃圾","衣物","浴室","公物","罰則","總分","等級","備註"]);
      sh.appendRow([payload.date,payload.b,payload.r,payload.ins,payload.fl,payload.dk,payload.bd,payload.tr,payload.cl,payload.bt,payload.cm,payload.pn,payload.tot,payload.gr,payload.note]);
    }

    if (type === "roll") {
      var sh = ss.getSheetByName("點名紀錄") || ss.insertSheet("點名紀錄");
      if (sh.getLastRow() === 0) sh.appendRow(["點名時間","棟別","房號","床位","姓名","狀態","班級"]);
      sh.appendRow([new Date().toLocaleString("zh-TW"),payload.building,payload.room,payload.bed,payload.name,payload.status,payload.cls]);
    }

    if (type === "rollBatch") {
      var sh = ss.getSheetByName("點名紀錄") || ss.insertSheet("點名紀錄");
      if (sh.getLastRow() === 0) sh.appendRow(["點名時間","棟別","房號","床位","姓名","狀態","班級"]);
      var rows2d = payload.rows.map(function(r){ return [r.time,r.building,r.room,r.bed,r.name,r.status,r.cls]; });
      if (rows2d.length) sh.getRange(sh.getLastRow()+1, 1, rows2d.length, 7).setValues(rows2d);
    }

    if (type === "writeStudents") {
      var sh = ss.getSheetByName("僑生資料") || ss.insertSheet("僑生資料");
      sh.clearContents();
      sh.appendRow(["學號","姓名","性別","年級","班級","國籍/地區","聯絡電話","緊急聯絡人","緊急聯絡電話","電子郵件","棟別","房號","床位","入住日期","住宿狀態","英文姓名","I僑卡號","先實習開始","先實習結束","後實習開始","後實習結束","輪調實習廠商"]);
      var rows2d = payload.map(function(s){ return [s.id,s.name,s.g,s.grade,s.cls,s.nation,s.phone,s.emgName,s.emgPhone,s.email,s.b,s.r,s.bed,s.inDate,s.st,s.eng,s.iCard,s.internStart1||"",s.internEnd1||"",s.internStart2||"",s.internEnd2||"",s.internCo||""]; });
      if (rows2d.length) sh.getRange(2, 1, rows2d.length, 22).setValues(rows2d);
    }

    if (type === "writeRooms") {
      var sh = ss.getSheetByName("房間設定") || ss.insertSheet("房間設定");
      sh.clearContents();
      sh.appendRow(["棟別","樓層","房號","房型","床位數","狀態"]);
      var rows2d = payload.map(function(r){ return [r.b,r.fl,r.r,r.t,r.beds,r.s]; });
      if (rows2d.length) sh.getRange(2, 1, rows2d.length, 6).setValues(rows2d);
    }

    return buildResp({ok:true});
  } catch(err) {
    return buildResp({ok:false, err:err.message});
  }
}

function doGet(e) {
  var param = e && e.parameter ? e.parameter : {};

  if (param.action === "getStudents") {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sh = ss.getSheetByName("僑生資料");
      if (!sh) return buildResp({ok:false, err:"找不到「僑生資料」工作表"});
      var rows = sh.getDataRange().getValues();
      var students = [];
      for (var i = 1; i < rows.length; i++) {
        var r = rows[i];
        if (!r[0]) continue;
        students.push({id:String(r[0]||""),name:String(r[1]||""),g:String(r[2]||""),grade:String(r[3]||"1"),cls:String(r[4]||""),nation:String(r[5]||""),phone:String(r[6]||""),emgName:String(r[7]||""),emgPhone:String(r[8]||""),email:String(r[9]||""),b:String(r[10]||""),r:String(r[11]||""),bed:String(r[12]||""),inDate:String(r[13]||""),st:String(r[14]||"入住"),eng:String(r[15]||""),iCard:String(r[16]||""),internStart1:String(r[17]||""),internEnd1:String(r[18]||""),internStart2:String(r[19]||""),internEnd2:String(r[20]||""),internCo:String(r[21]||"")});
      }
      return buildResp({ok:true, students:students});
    } catch(err) {
      return buildResp({ok:false, err:err.message});
    }
  }

  if (param.action === "getRooms") {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sh = ss.getSheetByName("房間設定");
      if (!sh) return buildResp({ok:false, err:"找不到「房間設定」工作表"});
      var rows = sh.getDataRange().getValues();
      var rooms = [];
      for (var i = 1; i < rows.length; i++) {
        var r = rows[i];
        if (!r[0]) continue;
        rooms.push({b:String(r[0]),fl:Number(r[1]||0),r:String(r[2]),t:String(r[3]),beds:Number(r[4]||6),s:String(r[5]||"正常")});
      }
      return buildResp({ok:true, rooms:rooms});
    } catch(err) {
      return buildResp({ok:false, err:err.message});
    }
  }

  return buildResp({ok:true, msg:"Connected", time:new Date().toLocaleString("zh-TW")});
}

function buildResp(obj) {
  var out = ContentService.createTextOutput(JSON.stringify(obj));
  out.setMimeType(ContentService.MimeType.JSON);
  return out;
}

5 執行身分：我　　誰可以存取：所有人（含匿名），按「部署」 → 第一次會要求授權，請允許
6 複製「網路應用程式 URL」，貼到上方欄位，按「💾 儲存設定」，再按「🔌 測試連線」確認 ✅
7 修改程式碼後需重新部署：「部署 → 管理部署」→「✏️ 編輯」→「版本：新版本」→「部署」並 更新 URL
⚠️ 常見連線失敗原因
❶ 「誰可以存取」未設為「所有人」 → 重新部署，選「所有人（含匿名）」
❷ 部署後沒更新 URL（舊版本失效）→ 複製最新的 Web App URL
❸ Apps Script 未授權執行 → 點部署後出現的授權視窗並允許
❹ 程式碼有修改但未重新部署 → 每次改程式都要建立新版本部署
